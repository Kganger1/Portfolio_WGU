CREATE TABLE tempibm (
    CustomerID VARCHAR(20) PRIMARY KEY,
    City VARCHAR(100),
    State CHAR(2),
    Zip_Code VARCHAR(10),
    Age INTEGER,
    Gender VARCHAR(10),
    Senior_Citizen VARCHAR(3),  -- 'Yes' or 'No'
    Marital VARCHAR(3),         -- 'Yes' or 'No'
    Dependents VARCHAR(3),      -- 'Yes' or 'No'
    Tenure_Months INTEGER,
    Phone VARCHAR(3),           -- 'Yes' or 'No'
    Multiple_Lines VARCHAR(3),  -- 'Yes' or 'No'
    Internet_Service VARCHAR(20),
    Online_Security VARCHAR(3), -- 'Yes' or 'No'
    Online_Backup VARCHAR(3),   -- 'Yes' or 'No'
    Device_Protection VARCHAR(3), -- 'Yes' or 'No'
    Tech_Support VARCHAR(3),    -- 'Yes' or 'No'
    Streaming_TV VARCHAR(3),    -- 'Yes' or 'No'
    Streaming_Movies VARCHAR(3), -- 'Yes' or 'No'
    Contract VARCHAR(20),
    Paperless_Billing VARCHAR(3), -- 'Yes' or 'No'
    Payment_Method VARCHAR(30),
    Monthly_Charges NUMERIC(5, 2), -- For charges like 53.85
    Churn VARCHAR(3)             -- 'Yes' or 'No'
);


UPDATE tempibm
SET Zip_Code = LPAD(Zip_Code, 5, '0');

UPDATE tempibm
SET internet_service = 'Fiber Optic'
WHERE internet_service = 'Fiber optic';

    customerid AS CustomerID,
    churn AS Churn,
    city AS City,
    state AS State,
    zip_code  AS ZipCode,
    age AS Age,
    gender AS Gender,
    dependents AS Children,
    marital AS Marital,
    phone AS Phone,
    multiple_lines AS MultipleLines,
    internet_service AS InternetService,

    device_protection AS DeviceProtection,
    tech_support AS TechSupport,
    streaming_tv AS StreamingTV,
    streaming_movies AS StreamingMovies,
    monthly_charges AS MonthlyCharges
